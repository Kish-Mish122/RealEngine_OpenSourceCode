@echo off
chcp 65001 >nul
title Real Engine - Fix .rlscene to .tres
color 0A

echo ============================================
echo   REAL ENGINE - ЗАМЕНА .rlscene НА .tres
echo ============================================
echo.

:: Переходим в папку с исходниками Godot
cd /d D:\Real_Engine\source_realengine

echo 1. Создаю бэкап всех изменяемых файлов...
set BACKUP_DIR=backup_fix_%date:~-4,4%%date:~-7,2%%date:~-10,2%_%time:~0,2%%time:~3,2%
mkdir "%BACKUP_DIR%" 2>nul
echo   ✓ Бэкап создан: %BACKUP_DIR%
echo.

echo 2. Заменяю .rlscene на .tres во всех .cpp и .h файлах...
echo.

:: Счётчики
set TOTAL_FILES=0
set TOTAL_CHANGES=0

:: Обрабатываем все .cpp и .h файлы
for /r . %%f in (*.cpp *.h) do (
    findstr /m "rlscene" "%%f" >nul 2>nul
    if !errorlevel! equ 0 (
        set /a TOTAL_FILES+=1
        echo   Обрабатываю: %%f
        
        :: Создаём копию в бэкапе
        set "rel_path=%%f"
        set "rel_path=!rel_path:%CD%\=!"
        set "backup_path=%BACKUP_DIR%\!rel_path!"
        for %%p in ("!backup_path!") do mkdir "%%~dpp" 2>nul
        copy "%%f" "!backup_path!" >nul
        
        :: Заменяем .rlscene на .tres
        powershell -Command "(Get-Content '%%f') -replace 'rlscene', 'tres' | Set-Content '%%f' -Encoding UTF8"
        
        :: Считаем количество замен
        powershell -Command "$c = (Select-String -Path '%%f' -Pattern 'tres' -AllMatches).Matches.Count; Write-Host $c" > %TEMP%\count.txt
        set /p CHANGES=<%TEMP%\count.txt
        set /a TOTAL_CHANGES+=!CHANGES!
        
        echo     ✓ Замен: !CHANGES!
    )
)

echo.
echo 3. Проверяю остальные файлы...
echo.

:: Обрабатываем .py и другие файлы
for /r . %%f in (*.py *.xml *.cfg) do (
    findstr /m "rlscene" "%%f" >nul 2>nul
    if !errorlevel! equ 0 (
        set /a TOTAL_FILES+=1
        echo   Обрабатываю: %%f
        
        :: Создаём копию в бэкапе
        set "rel_path=%%f"
        set "rel_path=!rel_path:%CD%\=!"
        set "backup_path=%BACKUP_DIR%\!rel_path!"
        for %%p in ("!backup_path!") do mkdir "%%~dpp" 2>nul
        copy "%%f" "!backup_path!" >nul
        
        :: Заменяем .rlscene на .tres
        powershell -Command "(Get-Content '%%f') -replace 'rlscene', 'tres' | Set-Content '%%f' -Encoding UTF8"
    )
)

echo.
echo ============================================
echo   РЕЗУЛЬТАТЫ
echo ============================================
echo.
echo   Изменено файлов: %TOTAL_FILES%
echo   Всего замен: %TOTAL_CHANGES%
echo   Бэкап: %BACKUP_DIR%
echo.

:: Проверка остатков
echo 4. Проверяю остаточные вхождения...
echo.
findstr /s /i "rlscene" *.cpp *.h *.py 2>nul
if !errorlevel! equ 0 (
    echo   ⚠ Найдены остаточные вхождения!
    echo   Список выше.
) else (
    echo   ✓ Все вхождения .rlscene заменены на .tres!
)

echo.
echo 5. Важно: Проверьте файл editor_settings.cpp вручную!
echo    Путь: editor/settings/editor_settings.cpp
echo.
echo   Если там есть rlscene, замените вручную или запустите:
echo   sed -i 's/rlscene/tres/g' editor/settings/editor_settings.cpp
echo.

echo Нажмите любую клавишу для выхода...
pause >nul
